function route(){
  const h = decodeURIComponent(location.hash.slice(1));
  const [key, qs] = h.split("?");
  if(key && !ROUTES[key]){
    // رابط داخلي لعنصر داخل الصفحة الحالية (مثل #paths)
    const el = document.getElementById(key);
    if(el){ if(!el.closest(".page.active")) showPage("home"); el.scrollIntoView({behavior:"smooth"}); }
    return;
  }
  const route = key || "home";
  const params = new URLSearchParams(qs||"");
  showPage(ROUTES[route]);
  $$("[data-nav]").forEach(a => { if(a.dataset.nav===route) a.setAttribute("aria-current","page"); else a.removeAttribute("aria-current"); });
  if(route==="grad" || route==="faculty"){
    svcState.track = route==="grad" ? "grad" : "fac";
    svcState.cat = params.get("cat") || "all";
    svcState.stage = params.get("stage") || "";
    svcState.q = "";
    $("#svcSearch").value = ""; $("#svcStage").value = svcState.stage;
    renderTrack();
  }
  if(route==="contact" && params.get("service")) $("#f-svc").value = params.get("service");
  const title = route==="grad"||route==="faculty" ? TRACKS[svcState.track].title : $("#page-"+ROUTES[route]).dataset.title;
  document.title = route==="home" ? `${CONFIG.teamName} | استشارات بحثية لطلاب الدراسات العليا وأعضاء هيئة التدريس` : `${title} | ${CONFIG.teamName}`;
  closeDrawer();
  window.scrollTo({top:0, behavior:"auto"});
}
function showPage(id){ $$(".page").forEach(p => p.classList.toggle("active", p.id === "page-"+id)); }

function groupHtml(c, items){

function renderServices(){
  $$(".chip").forEach(c => c.setAttribute("aria-pressed", c.dataset.cat===svcState.cat ? "true":"false"));
  const q = norm(svcState.q.trim());
  const list = SERVICES.filter(s => trackOf(s)===svcState.track && (svcState.cat==="all"||s.cat===svcState.cat) && (!svcState.stage||s.stages.includes(svcState.stage)) && (!q || norm(s.t+" "+s.d+" "+catById(s.cat).name).includes(q)));
  $("#svcCount").textContent = `${list.length} خدمة`;
  if(!list.length){
    $("#svcList").innerHTML = `<div class="empty"><h3 style="margin-bottom:.5rem">لم نجد خدمة بهذا الوصف</h3><p class="muted">اكتب لنا ما تحتاجه، فقد نقدّمه دون أن يكون مذكورًا هنا.</p><a class="btn btn-wa" href="${waLink("السلام عليكم، أحتاج إلى خدمة: "+svcState.q)}" target="_blank" rel="noopener">${svg("wa")}اسألنا عبر واتساب</a></div>`;
    return;
  }
  const cats = CATS.filter(c=>c.track===svcState.track);
  const main = cats.filter(c=>!c.support).map(c=>({c,items:list.filter(s=>s.cat===c.id)})).filter(g=>g.items.length);
  const sup  = cats.filter(c=>c.support).map(c=>({c,items:list.filter(s=>s.cat===c.id)})).filter(g=>g.items.length);
  let html = main.map(g=>groupHtml(g.c,g.items)).join("");
  if(sup.length){
    html += `<div class="support-divider"><h2>خدمات مساندة</h2></div><p class="support-note">خدمات تصميم وتقنية تكمّل عملك البحثي، يقدّمها متخصصون من الفريق.</p>` + sup.map(g=>groupHtml(g.c,g.items)).join("");
  }
  $("#svcList").innerHTML = html;
}

function initForm(){
  $("#f-stage").innerHTML = `<option value="">اختر</option>` + STAGES.map(s=>`<option>${s.name}</option>`).join("") + `<option>بحث لعضو هيئة تدريس</option>`;
  const opts = track => CATS.filter(c=>c.track===track).map(c => `<optgroup label="${c.name}">${SERVICES.filter(s=>s.cat===c.id).map(s=>`<option>${esc(s.t)}</option>`).join("")}</optgroup>`).join("");
  $("#f-svc").innerHTML = `<option value="">اختر الخدمة</option><option>لست متأكدًا، أحتاج إلى إرشاد</option>` + opts("grad") + opts("fac") + CONFIG.courses.map(c=>`<option>${esc(c.title)}</option>`).join("");
  const form = $("#reqForm"), msg = $("#formMsg");
  function fail(t, el){ msg.className="form-msg err"; msg.textContent=t; el && el.focus(); return false; }
  function validate(){
    if(!$("#f-name").value.trim()) return fail("اكتب اسمك.", $("#f-name"));
    if($("#f-wa").value.replace(/\D/g,"").length < 8) return fail("اكتب رقم واتساب صحيحًا مع رمز الدولة.", $("#f-wa"));
    if(!$("#f-svc").value) return fail("اختر الخدمة المطلوبة.", $("#f-svc"));
    if(!$("#f-consent").checked) return fail("وافق على شروط الخدمة وسياسة الخصوصية لإرسال الطلب.", $("#f-consent"));
    return true;
  }
  form.addEventListener("submit", async e => {
    e.preventDefault();
    if(!validate()) return;
    const btn = $("#formSubmit"); btn.disabled = true; btn.textContent = "جارٍ الإرسال…";
    try{
      const res = await fetch("/", {method:"POST", headers:{"Content-Type":"application/x-www-form-urlencoded"}, body:new URLSearchParams(new FormData(form)).toString()});
      if(!res.ok) throw new Error(res.status);
      msg.className="form-msg ok"; msg.textContent="وصلنا طلبك. سنتواصل معك على رقم الواتساب الذي كتبته.";
      form.reset();
    }catch(err){
      msg.className="form-msg err"; msg.textContent="تعذّر إرسال النموذج الآن. اضغط «أرسل البيانات نفسها عبر واتساب» ليصلنا طلبك فورًا.";
    }finally{ btn.disabled=false; btn.textContent="إرسال الطلب"; }
  });
  $("#formToWa").addEventListener("click", () => {
    const d = Object.fromEntries(new FormData(form).entries());
    const lines = ["السلام عليكم ورحمة الله،", `أرغب في خدمة: «${d.service||"لم أحدد بعد"}»`,
      d.name && `الاسم: ${d.name}`, d.role && `الصفة: ${d.role}`, d.stage && `المرحلة: ${d.stage}`,
      d.university && `الجامعة: ${d.university}`, d.specialty && `التخصص: ${d.specialty}`,
      d.deadline && `الموعد المطلوب: ${d.deadline}`, d.details && `التفاصيل: ${d.details}`].filter(Boolean).join("\n");
    window.open(waLink(lines), "_blank", "noopener");
  });
}
